function renderBonuses() { const block = document.getElementById('uiBonusesList'); // 1. Если сервер прислал массив bonuses — рисуем его if (Array.isArray(currentPlayer.bonuses) && currentPlayer.bonuses.length > 0) { block.innerHTML = currentPlayer.bonuses.map(b => renderBonusItem(b)).join(''); return; } // 2. Fallback: старый bonus_text / bonus_code if (currentPlayer.bonus_text || currentPlayer.bonus_code) { block.innerHTML = `
🎁 Ваш бонус
${currentPlayer.bonus_text || ''}
${currentPlayer.bonus_code ? `
${currentPlayer.bonus_code}
` : ''}
`; return; } // 3. Fallback: если есть бонусный баланс, но нет массива — показываем как «бонус за ранг» const bonusBal = parseFloat(currentPlayer.bonus_balance || 0); if (bonusBal > 0) { block.innerHTML = renderBonusItem({ id: 'rank_bonus_' + (currentPlayer.rank || 'Bronze'), type: 'rank', title: 'Бонус за повышение ранга', desc: 'Начислен за достижение ранга ' + (currentPlayer.rank || 'Bronze'), amount: bonusBal, can_refuse: true, status: 'active', wager: 'x3' }); return; } block.innerHTML = `
Нет активных бонусов
`; } function renderBonusItem(b) { const rejected = b.status === 'rejected'; const iconClass = b.type === 'rank' ? 'rank' : 'regular'; const iconEmoji = b.type === 'rank' ? '🏆' : b.type === 'deposit' ? '💳' : b.type === 'cashback' ? '↩️' : '🎁'; const itemClass = 'bonus-item' + (b.type === 'rank' ? ' rank-bonus' : '') + (rejected ? ' rejected' : ''); const sumClass = rejected ? 'bonus-item-sum rejected-sum' : 'bonus-item-sum'; const tags = rejected ? 'Отклонён' : (b.wager ? `Вейджер ${b.wager}` : '') + 'Активен'; const btn = (b.can_refuse && !rejected) ? `` : ''; return `
${iconEmoji}
${b.title || 'Бонус'}
${b.desc || ''}
${tags}
${parseFloat(b.amount).toFixed(2)} ₽
${btn}
`; } function openRefuseBonusModal(bonusId, amount) { pendingRefuseBonusId = bonusId; document.getElementById('refuseBonusText').textContent = `Бонус на ${parseFloat(amount).toFixed(2)} ₽ будет списан с бонусного баланса. Это действие нельзя отменить.`; openModal('refuseBonusModal'); } async function confirmRefuseBonus() { if (!pendingRefuseBonusId) return; const bonusId = pendingRefuseBonusId; const btn = document.getElementById('refuseBonusConfirmBtn'); btn.disabled = true; btn.textContent = '...'; try { const res = await fetch('/api/guest/refuse-bonus', { method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify({ id: currentPlayer.id, bonus_id: bonusId }) }); const data = await res.json().catch(() => ({})); if (!res.ok) { showToast(data.error || 'Не удалось отказаться', 'error'); btn.disabled = false; btn.textContent = 'Отказаться'; return; } showToast('Бонус отклонён', 'success'); closeModal('refuseBonusModal'); pendingRefuseBonusId = null; btn.disabled = false; btn.textContent = 'Отказаться'; loadProfile(); // перезагрузить баланс и бонусы } catch(e) { showToast('Ошибка соединения', 'error'); btn.disabled = false; btn.textContent = 'Отказаться'; } }
